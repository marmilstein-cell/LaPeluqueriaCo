    36→/** Routea el hash #cabina — igual patrón que #session/BRD-XXXX. */
    37→function useCabinaRoute() {
    38→  const [open, setOpen] = useState(false);
    39→  useEffect(() => {
    40→    const parse = () => setOpen(window.location.hash.toLowerCase() === "#cabina");
    41→    parse();
    42→    window.addEventListener("hashchange", parse);
    43→    return () => window.removeEventListener("hashchange", parse);
    44→  }, []);
    45→  return open;
    46→}
    47→
    48→export function Cabina() {
    49→  const routeOpen = useCabinaRoute();
    50→  // código persistido en sessionStorage (la cabina es de uso diario);
    51→  // lazy initializer: el overlay nunca se renderiza server-side
    52→  const [code, setCode] = useState<string | null>(() => {
    53→    if (typeof window === "undefined") return null;
    54→    try {
    55→      return sessionStorage.getItem(CABINA_CODE_KEY);
    56→    } catch {
    57→      return null;
    58→    }
    59→  });
    60→  const queryClient = useQueryClient();
    61→  const overlayRef = useRef<HTMLDivElement>(null);
    62→  // la agenda arranca en hoy (hora real de Buenos Aires)
    63→  const [date, setDate] = useState(() => nowBuenosAires().iso);
    64→
    65→  const close = useCallback(() => {
    66→    if (window.location.hash.toLowerCase() === "#cabina") {
    67→      window.history.pushState(null, "", "#book");
    68→    }
    69→    window.dispatchEvent(new HashChangeEvent("hashchange"));
    70→    setTimeout(() => scrollToScene("book"), 60);
    71→  }, []);
    72→
    73→  // congelar el scroll del sitio mientras la cabina está abierta
    74→  useEffect(() => {
    75→    if (routeOpen) {
    76→      const lenis = (window as Window & { __lenis?: { stop(): void; start(): void } }).__lenis;
    77→      lenis?.stop();
    78→      document.documentElement.style.overflow = "hidden";
    79→      // la cabina toma la pantalla: si había un pase abierto, se guarda
    80→      // (queda en su URL #session/ID y en el historial local).
    81→      useBooking.setState({ session: null });
    82→    } else {
    83→      const lenis = (window as Window & { __lenis?: { stop(): void; start(): void } }).__lenis;
    84→      lenis?.start();
    85→      document.documentElement.style.overflow = "";
    86→    }
    87→  }, [routeOpen]);
    88→
    89→  useFocusTrap(routeOpen, overlayRef, { onEsc: close });
    90→
    91→  return (
    92→    <AnimatePresence>
    93→      {routeOpen && (
    94→        <motion.div
    95→          key="cabina"
    96→          ref={overlayRef}
    97→          role="dialog"
    98→          aria-modal="true"
    99→          aria-label="La Cabina — gestión interna de Bardos"
   100→          className="fixed inset-0 z-[80] overflow-y-auto bg-black"
   101→          initial={{ opacity: 0 }}
   102→          animate={{ opacity: 1 }}
   103→          exit={{ opacity: 0 }}
   104→          transition={{ duration: 0.35 }}
   105→        >
   106→          <div className="mx-auto w-full max-w-6xl px-4 pb-24 pt-20 md:px-8 md:pt-24">
   107→            {!code ? (
   108→              <CabinaGate
   109→                onValid={(c) => {
   110→                  sessionStorage.setItem(CABINA_CODE_KEY, c);
   111→                  setCode(c);
   112→                }}
   113→                onClose={close}
   114→              />
   115→            ) : (
   116→              <CabinaDashboard
   117→                code={code}
   118→                date={date}
   119→                setDate={setDate}
   120→                onLogout={() => {
   121→                  sessionStorage.removeItem(CABINA_CODE_KEY);
   122→                  setCode(null);
   123→                  queryClient.clear();
   124→                }}
   125→                onClose={close}
   126→              />
   127→            )}
   128→          </div>
   129→        </motion.div>
   130→      )}
   131→    </AnimatePresence>
   132→  );
   133→}

   135→/* ---------------- GATE: el código de la cabina ---------------- */
   136→
   137→function CabinaGate({
   138→  onValid,
   139→  onClose,
   140→}: {
   141→  onValid: (code: string) => void;
   142→  onClose: () => void;
   143→}) {
   144→  const [value, setValue] = useState("");
   145→  const [error, setError] = useState<string | null>(null);
   146→  const [checking, setChecking] = useState(false);
   147→
   148→  const submit = async (e: React.FormEvent) => {
   149→    e.preventDefault();
   150→    const candidate = value.trim().toUpperCase();
   151→    if (candidate.length < 3) {
   152→      setError("El código es más largo.");
   153→      return;
   154→    }
   155→    setChecking(true);
   156→    setError(null);
   157→    try {
   158→      // validación real: la API devuelve 401 si el código no es
   159→      await fetchAgenda(candidate, nowBuenosAires().iso);
   160→      onValid(candidate);
   161→    } catch {
   162→      setError("Código cortado. Fijate de nuevo.");
   163→    } finally {
   164→      setChecking(false);
   165→    }
   166→  };
   167→
   168→  return (
   169→    <motion.div
   170→      className="mx-auto flex min-h-[60vh] max-w-md flex-col justify-center"
   171→      initial={{ opacity: 0, y: 24 }}
   172→      animate={{ opacity: 1, y: 0 }}
   173→      transition={{ duration: 0.5, ease: [0.16, 1, 0.3, 1] }}
   174→    >
   175→      <p className="type-micro text-smoke">BARDOS / SOLO PERSONAL</p>
   176→      <h2 className="font-display font-display-tight type-display-md mt-4 text-offwhite">
   177→        LA CABINA<span className="text-red">.</span>
   178→      </h2>
   179→      <p className="font-editorial-italic mt-4 text-[clamp(17px,1.8vw,22px)] text-chalk/85">
   180→        Acá se ve el día antes que el cliente.
   181→      </p>
   182→
   183→      <form onSubmit={submit} className="mt-10">
   184→        <label htmlFor="cabina-code" className="type-micro block text-bone">
   185→          CÓDIGO DE ACCESO
   186→        </label>
   187→        <input
   188→          id="cabina-code"
   189→          value={value}
   190→          onChange={(e) => {
   191→            setValue(e.target.value.toUpperCase());
   192→            setError(null);
   193→          }}
   194→          autoFocus
   195→          autoComplete="off"
   196→          spellCheck={false}
   197→          disabled={checking}
   198→          placeholder="••••••"
   199→          className="b-input mt-2 font-display text-[clamp(22px,3vw,34px)] tracking-[0.32em]"
   200→          style={{ textTransform: "uppercase" }}
   201→        />
   202→        {error && <p className="type-micro mt-3 text-red">{error}</p>}
   203→        <div className="mt-6 flex items-center gap-6">
   204→          <button type="submit" disabled={checking} className="b-btn">
   205→            {checking ? "ABRIENDO…" : "ENTRAR"}
   206→          </button>
   207→          <button type="button" onClick={onClose} className="b-link">
   208→            VOLVER AL SITIO
   209→          </button>
   210→        </div>
   211→      </form>
   212→    </motion.div>
   213→  );
   214→}

   502→/* ---------------- TIMELINE: 10:00→20:00 en slots de 30′ ---------------- */
   503→
   504→function AgendaTimeline({
   505→  artist,
   506→  date,
   507→  onBlock,
   508→  onUnblock,
   509→  onComplete,
   510→  onReopen,
   511→  busy,
   512→  error,
   513→}: {
   514→  artist: AgendaArtist;
   515→  date: string;
   516→  onBlock: (time: string, reason: string | undefined, minutes: number) => void;
   517→  onUnblock: (id: string) => void;
   518→  onComplete: (id: string) => void;
   519→  onReopen: (id: string) => void;
   520→  busy: boolean;
   521→  error: string | null;
   522→}) {
   523→  const [pending, setPending] = useState<string | null>(null); // slot en dialog de bloqueo
   524→  const [reason, setReason] = useState("");
   525→  const [duration, setDuration] = useState<number>(SHOP.slotStep);
   526→  const [confirmUnblock, setConfirmUnblock] = useState<string | null>(null);
   527→
   528→  // reloj vivo (30″): la línea AHORA y el sombreado del pasado se mueven solos
   529→  const [clock, setClock] = useState(() => nowBuenosAires());
   530→  useEffect(() => {
   531→    const id = setInterval(() => setClock(nowBuenosAires()), 30_000);
   532→    return () => clearInterval(id);
   533→  }, []);
   534→  const isToday = clock.iso === date;
   535→
   536→  // grilla completa: slots de 30' desde openHour hasta closeHour-slotStep
   537→  const slots = useMemo(() => {
   538→    const out: string[] = [];
   539→    for (let t = SHOP.openHour * 60; t < SHOP.closeHour * 60; t += SHOP.slotStep) {
   540→      out.push(minutesToTime(t));
   541→    }
   542→    return out;
   543→  }, []);
   544→
   545→  // bookings y blocks posicionados por índice de slot
   546→  const bookingsByStart = new Map(artist.bookings.map((b) => [b.time, b]));

   839→              </div>

   842→
   843→          {/* línea de AHORA: el corte rojo que avanza con el día */}

   875→                    disabled={busy}

   883→                )}
   884→              </li>
   885→            ))}
   886→          </ul>
   887→        </div>
   888→      )}
   889→    </div>
   890→  );
   891→}
