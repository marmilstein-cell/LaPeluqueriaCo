    36→    36→/** Routea el hash #cabina — igual patrón que #session/BRD-XXXX. */
    37→    37→function useCabinaRoute() {
    38→    38→  const [open, setOpen] = useState(false);
    39→    39→  useEffect(() => {
    40→    40→    const parse = () => setOpen(window.location.hash.toLowerCase() === "#cabina");
    41→    41→    parse();
    42→    42→    window.addEventListener("hashchange", parse);
    43→    43→    return () => window.removeEventListener("hashchange", parse);
    44→    44→  }, []);
    45→    45→  return open;
    46→    46→}
    47→    47→
    48→    48→export function Cabina() {
    49→    49→  const routeOpen = useCabinaRoute();
    50→    50→  // código persistido en sessionStorage (la cabina es de uso diario);
    51→    51→  // lazy initializer: el overlay nunca se renderiza server-side
    52→    52→  const [code, setCode] = useState<string | null>(() => {
    53→    53→    if (typeof window === "undefined") return null;
    54→    54→    try {
    55→    55→      return sessionStorage.getItem(CABINA_CODE_KEY);
    56→    56→    } catch {
    57→    57→      return null;
    58→    58→    }
    59→    59→  });
    60→    60→  const queryClient = useQueryClient();
    61→    61→  const overlayRef = useRef<HTMLDivElement>(null);
    62→    62→  // la agenda arranca en hoy (hora real de Buenos Aires)
    63→    63→  const [date, setDate] = useState(() => nowBuenosAires().iso);
    64→    64→
    65→    65→  const close = useCallback(() => {
    66→    66→    if (window.location.hash.toLowerCase() === "#cabina") {
    67→    67→      window.history.pushState(null, "", "#book");
    68→    68→    }
    69→    69→    window.dispatchEvent(new HashChangeEvent("hashchange"));
    70→    70→    setTimeout(() => scrollToScene("book"), 60);
    71→    71→  }, []);
    72→    72→
    73→    73→  // congelar el scroll del sitio mientras la cabina está abierta
    74→    74→  useEffect(() => {
    75→    75→    if (routeOpen) {
    76→    76→      const lenis = (window as Window & { __lenis?: { stop(): void; start(): void } }).__lenis;
    77→    77→      lenis?.stop();
    78→    78→      document.documentElement.style.overflow = "hidden";
    79→    79→      // la cabina toma la pantalla: si había un pase abierto, se guarda
    80→    80→      // (queda en su URL #session/ID y en el historial local).
    81→    81→      useBooking.setState({ session: null });
    82→    82→    } else {
    83→    83→      const lenis = (window as Window & { __lenis?: { stop(): void; start(): void } }).__lenis;
    84→    84→      lenis?.start();
    85→    85→      document.documentElement.style.overflow = "";
    86→    86→    }
    87→    87→  }, [routeOpen]);
    88→    88→
    89→    89→  useFocusTrap(routeOpen, overlayRef, { onEsc: close });
    90→    90→
    91→    91→  return (
    92→    92→    <AnimatePresence>
    93→    93→      {routeOpen && (
    94→    94→        <motion.div
    95→    95→          key="cabina"
    96→    96→          ref={overlayRef}
    97→    97→          role="dialog"
    98→    98→          aria-modal="true"
    99→    99→          aria-label="La Cabina — gestión interna de Bardos"
   100→   100→          className="fixed inset-0 z-[80] overflow-y-auto bg-black"
   101→   101→          initial={{ opacity: 0 }}
   102→   102→          animate={{ opacity: 1 }}
   103→   103→          exit={{ opacity: 0 }}
   104→   104→          transition={{ duration: 0.35 }}
   105→   105→        >
   106→   106→          <div className="mx-auto w-full max-w-6xl px-4 pb-24 pt-20 md:px-8 md:pt-24">
   107→   107→            {!code ? (
   108→   108→              <CabinaGate
   109→   109→                onValid={(c) => {
   110→   110→                  sessionStorage.setItem(CABINA_CODE_KEY, c);
   111→   111→                  setCode(c);
   112→   112→                }}
   113→   113→                onClose={close}
   114→   114→              />
   115→   115→            ) : (
   116→   116→              <CabinaDashboard
   117→   117→                code={code}
   118→   118→                date={date}
   119→   119→                setDate={setDate}
   120→   120→                onLogout={() => {
   121→   121→                  sessionStorage.removeItem(CABINA_CODE_KEY);
   122→   122→                  setCode(null);
   123→   123→                  queryClient.clear();
   124→   124→                }}
   125→   125→                onClose={close}
   126→   126→              />
   127→   127→            )}
   128→   128→          </div>
   129→   129→        </motion.div>
   130→   130→      )}
   131→   131→    </AnimatePresence>
   132→   132→  );
   133→   133→}

   135→   135→/* ---------------- GATE: el código de la cabina ---------------- */
   136→   136→
   137→   137→function CabinaGate({
   138→   138→  onValid,
   139→   139→  onClose,
   140→   140→}: {
   141→   141→  onValid: (code: string) => void;
   142→   142→  onClose: () => void;
   143→   143→}) {
   144→   144→  const [value, setValue] = useState("");
   145→   145→  const [error, setError] = useState<string | null>(null);
   146→   146→  const [checking, setChecking] = useState(false);
   147→   147→
   148→   148→  const submit = async (e: React.FormEvent) => {
   149→   149→    e.preventDefault();
   150→   150→    const candidate = value.trim().toUpperCase();
   151→   151→    if (candidate.length < 3) {
   152→   152→      setError("El código es más largo.");
   153→   153→      return;
   154→   154→    }
   155→   155→    setChecking(true);
   156→   156→    setError(null);
   157→   157→    try {
   158→   158→      // validación real: la API devuelve 401 si el código no es
   159→   159→      await fetchAgenda(candidate, nowBuenosAires().iso);
   160→   160→      onValid(candidate);
   161→   161→    } catch {
   162→   162→      setError("Código cortado. Fijate de nuevo.");
   163→   163→    } finally {
   164→   164→      setChecking(false);
   165→   165→    }
   166→   166→  };
   167→   167→
   168→   168→  return (
   169→   169→    <motion.div
   170→   170→      className="mx-auto flex min-h-[60vh] max-w-md flex-col justify-center"
   171→   171→      initial={{ opacity: 0, y: 24 }}
   172→   172→      animate={{ opacity: 1, y: 0 }}
   173→   173→      transition={{ duration: 0.5, ease: [0.16, 1, 0.3, 1] }}
   174→   174→    >
   175→   175→      <p className="type-micro text-smoke">BARDOS / SOLO PERSONAL</p>
   176→   176→      <h2 className="font-display font-display-tight type-display-md mt-4 text-offwhite">
   177→   177→        LA CABINA<span className="text-red">.</span>
   178→   178→      </h2>
   179→   179→      <p className="font-editorial-italic mt-4 text-[clamp(17px,1.8vw,22px)] text-chalk/85">
   180→   180→        Acá se ve el día antes que el cliente.
   181→   181→      </p>
   182→   182→
   183→   183→      <form onSubmit={submit} className="mt-10">
   184→   184→        <label htmlFor="cabina-code" className="type-micro block text-bone">
   185→   185→          CÓDIGO DE ACCESO
   186→   186→        </label>
   187→   187→        <input
   188→   188→          id="cabina-code"
   189→   189→          value={value}
   190→   190→          onChange={(e) => {
   191→   191→            setValue(e.target.value.toUpperCase());
   192→   192→            setError(null);
   193→   193→          }}
   194→   194→          autoFocus
   195→   195→          autoComplete="off"
   196→   196→          spellCheck={false}
   197→   197→          disabled={checking}
   198→   198→          placeholder="••••••"
   199→   199→          className="b-input mt-2 font-display text-[clamp(22px,3vw,34px)] tracking-[0.32em]"
   200→   200→          style={{ textTransform: "uppercase" }}
   201→   201→        />
   202→   202→        {error && <p className="type-micro mt-3 text-red">{error}</p>}
   203→   203→        <div className="mt-6 flex items-center gap-6">
   204→   204→          <button type="submit" disabled={checking} className="b-btn">
   205→   205→            {checking ? "ABRIENDO…" : "ENTRAR"}
   206→   206→          </button>
   207→   207→          <button type="button" onClick={onClose} className="b-link">
   208→   208→            VOLVER AL SITIO
   209→   209→          </button>
   210→   210→        </div>
   211→   211→      </form>
   212→   212→    </motion.div>
   213→   213→  );
   214→   214→}

   216→   216→/* ---------------- DASHBOARD: agenda + bloqueos + stats ---------------- */
   217→   217→
   218→   218→function CabinaDashboard({
   219→   219→  code,
   220→   220→  date,
   221→   221→  setDate,
   222→   222→  onLogout,
   223→   223→  onClose,
   224→   224→}: {
   225→   225→  code: string;
   226→   226→  date: string;
   227→   227→  setDate: (d: string) => void;
   228→   228→  onLogout: () => void;
   229→   229→  onClose: () => void;
   230→   230→}) {
   231→   231→  const [artistId, setArtistId] = useState<string | null>(null);
   232→   232→  const qc = useQueryClient();
   233→   233→  const today = useMemo(() => nowBuenosAires().iso, []);
   234→   234→
   235→   235→  const { data, isLoading, isError, refetch } = useQuery({
   236→   236→    queryKey: ["cabina-agenda", code, date],
   237→   237→    queryFn: () => fetchAgenda(code, date),
   238→   238→    enabled: !!date, // date se setea al montar (hoy BA) — evita un fetch inválido
   239→   239→    refetchInterval: 30_000,
   240→   240→  });
   241→   241→
   242→   242→  // artista activo: selección explícita o el primero del catálogo (estado derivado)
   243→   243→  const activeArtistId =
   244→   244→    artistId && data?.artists.some((a) => a.id === artistId)
   245→   245→      ? artistId
   246→   246→      : (data?.artists[0]?.id ?? null);
   247→   247→  const artist = data?.artists.find((a) => a.id === activeArtistId) ?? null;
   248→   248→  const stats = data?.stats;
   249→   249→
   250→   250→  const blockMut = useMutation({
   251→   251→    mutationFn: (input: {
   252→   252→      artistId: string;
   253→   253→      date: string;
   254→   254→      time: string;
   255→   255→      minutes?: number;
   256→   256→      reason?: string;
   257→   257→    }) => createBlock(code, input),
   258→   258→    onSuccess: () => {
   259→   259→      qc.invalidateQueries({ queryKey: ["cabina-agenda"] });
   260→   260→      // también la disponibilidad pública cambió
   261→   261→      qc.invalidateQueries({ queryKey: ["availability"] });
   262→   262→    },
   263→   263→  });
   264→   264→
   265→   265→  const unblockMut = useMutation({
   266→   266→    mutationFn: (id: string) => deleteBlock(code, id),
   267→   267→    onSuccess: () => {
   268→   268→      qc.invalidateQueries({ queryKey: ["cabina-agenda"] });
   269→   269→      qc.invalidateQueries({ queryKey: ["availability"] });
   270→   270→    },
   271→   271→  });
   272→   272→
   273→   273→  // ciclo de vida de la sesión: confirmada → completada (el corte está hecho)
   274→   274→  const statusMut = useMutation({
   275→   275→    mutationFn: (input: { id: string; status: "completed" | "confirmed" }) =>
   276→   276→      setSessionStatus(code, input.id, input.status),
   277→   277→    onSuccess: () => {
   278→   278→      qc.invalidateQueries({ queryKey: ["cabina-agenda"] });
   279→   279→    },
   280→   280→  });
   281→   281→
   282→   282→  return (
   283→   283→    <div>
   284→   284→      {/* claqueta */}
   285→   285→      <div className="flex flex-wrap items-center justify-between gap-4">
   286→   286→        <p className="type-micro text-smoke">
   287→   287→          BARDOS / CABINA — {formatLongDate(date || today)}
   288→   288→          {date === today && <span className="text-red"> — HOY</span>}
   289→   289→        </p>
   290→   290→        <div className="flex items-center gap-6">
   291→   291→          <button
   292→   292→            onClick={() => window.print()}
   293→   293→            className="b-link"
   294→   294→            title="Imprimir la hoja de ruta del día"
   295→   295→          >
   296→   296→            IMPRIMIR EL DÍA
   297→   297→          </button>
   298→   298→          <button
   299→   299→            onClick={onLogout}
   300→   300→            className="b-link"
   301→   301→            title="Salir de la cabina (borra el código de esta sesión)"
   302→   302→          >
   303→   303→            SALIR
   304→   304→          </button>
   305→   305→          <button onClick={onClose} className="b-link">
   306→   306→            CLOSE ×
   307→   307→          </button>
   308→   308→        </div>
   309→   309→      </div>
   310→   310→
   311→   311→      {/* hoja de ruta imprimible (portal: solo visible en @media print) */}
   312→   312→      <PrintSheet data={data ?? null} date={date} today={today} />
   313→   313→
   314→   314→      <div className="mt-6 flex flex-wrap items-end justify-between gap-6">
   315→   315→        <h2 className="font-display font-display-tight type-display-sm text-offwhite">
   316→   316→          LA CABINA<span className="text-red">.</span>
   317→   317→        </h2>
   318→   318→
   319→   319→        {/* picker de día — tira editorial como el builder */}
   320→   320→        {data && (
   321→   321→          <div
   322→   322→            className="-order-1 flex w-full gap-1 overflow-x-auto pb-2 md:order-none md:w-auto"
   323→   323→            style={{ scrollbarWidth: "thin" }}
   324→   324→            aria-label="Elegir día"
   325→   325→          >
   326→   326→            {data.dates.slice(0, 10).map((d) => {
   327→   327→              const l = formatDayLabel(d);
   328→   328→              const isSel = d === date;
   329→   329→              return (
   330→   330→                <button
   331→   331→                  key={d}
   332→   332→                  onClick={() => setDate(d)}
   333→   333→                  aria-pressed={isSel}
   334→   334→                  className={`group flex w-[64px] shrink-0 flex-col items-center border py-2 transition-colors duration-200 ${
   335→   335→                    isSel
   336→   336→                      ? "border-red text-offwhite"
   337→   337→                      : "border-line text-bone hover:border-smoke hover:text-chalk"
   338→   338→                  }`}
   339→   339→                >
   340→   340→                  <span className="type-micro text-[10px]">{l.day}</span>
   341→   341→                  <span className={`font-display text-[18px] ${isSel ? "text-red" : ""}`}>{l.num}</span>
   342→   342→                  <span className="type-micro text-[9px] text-smoke">
   343→   343→                    {d === today ? "HOY" : l.month}
   344→   344→                  </span>
   345→   345→                </button>
   346→   346→              );
   347→   347→            })}
   348→   348→          </div>
   349→   349→        )}
   350→   350→      </div>
   351→   351→
   352→   352→      {/* tabs de artista */}
   353→   353→      {data && (
   354→   354→        <div
   355→   355→          className="mt-8 flex flex-wrap gap-x-8 gap-y-3 border-b border-line pb-4"
   356→   356→          role="tablist"
   357→   357→          aria-label="Artistas"
   358→   358→        >
   359→   359→          {data.artists.map((a) => (
   360→   360→            <button
   361→   361→              key={a.id}
   362→   362→              role="tab"
   363→   363→              aria-selected={a.id === activeArtistId}
   364→   364→              onClick={() => setArtistId(a.id)}
   365→   365→              className={`font-display text-[clamp(16px,1.8vw,24px)] tracking-[0.04em] transition-colors duration-200 ${
   366→   366→                a.id === activeArtistId
   367→   367→                  ? "text-offwhite"
   368→   368→                  : "text-smoke hover:text-chalk"
   369→   369→              }`}
   370→   370→            >
   371→   371→              {a.name}
   372→   372→              <span
   373→   373→                className={`ml-2 inline-block h-1 w-1 rounded-full ${
   374→   374→                  a.id === activeArtistId ? "bg-red" : "bg-smoke/40"
   375→   375→                }`}
   376→   376→                aria-hidden="true"
   377→   377→              />
   378→   378→            </button>
   379→   379→          ))}
   380→   380→        </div>
   381→   381→      )}
   382→   382→
   383→   383→      {/* cuerpo: agenda + stats */}
   384→   384→      {isLoading && (
   385→   385→        <div className="flex flex-col items-center gap-6 py-24" role="status">
   386→   386→          <span className="cut-line w-20 animate-[pulse-line_1.6s_ease-in-out_infinite] origin-center" />
   387→   387→          <p className="type-micro text-bone">CARGANDO EL DÍA…</p>
   388→   388→        </div>
   389→   389→      )}
   390→   390→
   391→   391→      {isError && (
   392→   392→        <div className="border border-line bg-coal p-8 text-center">
   393→   393→          <p className="font-display text-[clamp(22px,3vw,36px)] text-offwhite">
   394→   394→            EL DÍA SE CORTÓ<span className="text-red">.</span>
   395→   395→          </p>
   396→   396→          <p className="font-system mt-3 text-[13px] text-bone/80">
   397→   397→            No pudimos traer la agenda. Puede ser el código o la conexión.
   398→   398→          </p>
   399→   399→          <button onClick={() => refetch()} className="b-btn b-btn-ghost mt-6">
   400→   400→            REINTENTAR
   401→   401→          </button>
   402→   402→        </div>
   403→   403→      )}
   404→   404→
   405→   405→      {data && artist && (
   406→   406→        <div className="mt-8 grid gap-10 lg:grid-cols-12">
   407→   407→          {/* agenda */}

   410→   410→              artist={artist}
   411→   411→              date={date}
   412→   412→              onBlock={(time, reason, minutes) =>
   413→   413→                blockMut.mutate({ artistId: artist.id, date, time, reason, minutes })
   414→   414→              }
   415→   415→              onUnblock={(id) => unblockMut.mutate(id)}
   416→   416→              onComplete={(id) => statusMut.mutate({ id, status: "completed" })}
   417→   417→              onReopen={(id) => statusMut.mutate({ id, status: "confirmed" })}
   418→   418→              busy={blockMut.isPending || unblockMut.isPending || statusMut.isPending}
   419→   419→              error={
   420→   420→                blockMut.error?.message ??
   421→   421→                unblockMut.error?.message ??
   422→   422→                statusMut.error?.message ??
   423→   423→                null
   424→   424→              }
   425→   425→            />
   426→   426→          </div>
   427→   427→
   428→   428→          {/* stats + clientes + ayuda */}
   429→   429→          <aside className="lg:col-span-4">
   430→   430→            {stats && (
   431→   431→              <div className="border border-line bg-coal">
   432→   432→                <p className="type-micro border-b border-line px-5 py-3 text-red">
   433→   433→                  EL DÍA EN NÚMEROS
   434→   434→                </p>
   435→   435→                <dl className="divide-y divide-line">
   436→   436→                  <Stat label="SESIONES CONFIRMADAS" value={String(stats.totalBookings)} />

   438→   438→                    label="CORTES COMPLETADOS"
   439→   439→                    value={`${stats.completed} / ${stats.totalBookings}`}
   440→   440→                  />
   441→   441→                  <Stat label="CAJA REALIZADA" value={priceARS(stats.revenueDone)} />

   443→   443→                  <Stat label="MINUTOS BLOQUEADOS" value={String(stats.totalBlockMinutes)} />
   444→   444→                  <Stat
   445→   445→                    label="OCUPACIÓN"
   446→   446→                    value={
   447→   447→                      stats.capacityMinutes > 0
   448→   448→                        ? `${Math.round((stats.bookedMinutes / stats.capacityMinutes) * 100)}%`
   449→   449→                        : "—"
   450→   450→                    }
   451→   451→                  />
   452→   452→                </dl>
   453→   453→              </div>
   454→   454→            )}
   455→   455→
   456→   456→            {/* búsqueda de clientes: quién llama */}
   457→   457→            <ClientLookup code={code} />

   506→   506→  date,

   509→   509→  onComplete,

   517→   517→  onUnblock: (id: string) => void;

   524→   524→  const [reason, setReason] = useState("");

   526→   526→  const [confirmUnblock, setConfirmUnblock] = useState<string | null>(null);
   527→   527→
   528→   528→  // reloj vivo (30″): la línea AHORA y el sombreado del pasado se mueven solos
   529→   529→  const [clock, setClock] = useState(() => nowBuenosAires());
   530→   530→  useEffect(() => {
   531→   531→    const id = setInterval(() => setClock(nowBuenosAires()), 30_000);
   532→   532→    return () => clearInterval(id);
   533→   533→  }, []);
   534→   534→  const isToday = clock.iso === date;
   535→   535→

   549→   549→
   550→   550→  // línea de AHORA: posición y visibilidad
   551→   551→  const nowTop = ((clock.minutes - SHOP.openHour * 60) / SHOP.slotStep) * ROW_H;
   552→   552→  const showNow =
   553→   553→    isToday &&
   554→   554→    clock.minutes >= SHOP.openHour * 60 &&
   555→   555→    clock.minutes <= SHOP.closeHour * 60;

   557→   557→  const emptyDay = artist.bookings.length === 0 && artist.blocks.length === 0;

   559→   559→  // duraciones que cierran dentro del día (nunca ofrecer un bloqueo imposible)
   560→   560→  const durationsFor = (time: string) => {
   561→   561→    const room = SHOP.closeHour * 60 - timeToMinutes(time);
   562→   562→    return BLOCK_DURATIONS.filter((m) => m <= room);
   563→   563→  };

   574→   574→        </p>
   575→   575→        <p className="type-micro text-smoke">
   576→   576→          {artist.bookings.length} SESIONES ·{" "}
   577→   577→          {artist.blocks.reduce((sum, bl) => sum + blockMinutes(bl.minutes), 0) / SHOP.slotStep}{" "}
   578→   578→          SLOTS BLOQUEADOS
   579→   579→        </p>
   580→   580→      </div>
   581→   581→
   582→   582→      {/* leyenda de lectura rápida */}
   583→   583→      <div className="mb-5 flex flex-wrap items-center gap-x-6 gap-y-2" aria-hidden="true">
   584→   584→        <span className="flex items-center gap-2 type-micro text-smoke">
   585→   585→          <span
   586→   586→            className="h-2.5 w-2.5 bg-coal"
   587→   587→            style={{ border: "1px solid var(--color-line)", borderLeft: "3px solid var(--color-red)" }}
   588→   588→          />
   589→   589→          SESIÓN
   590→   590→        </span>
   591→   591→        <span className="flex items-center gap-2 type-micro text-smoke">
   592→   592→          <span
   593→   593→            className="h-2.5 w-2.5"
   594→   594→            style={{
   595→   595→              background: "repeating-linear-gradient(-45deg, rgba(229,35,27,0.35) 0 3px, rgba(229,35,27,0.08) 3px 6px)",
   596→   596→              border: "1px solid rgba(229,35,27,0.5)",
   597→   597→            }}
   598→   598→          />
   599→   599→          BLOQUEO
   600→   600→        </span>
   601→   601→        <span className="flex items-center gap-2 type-micro text-smoke">
   602→   602→          <span className="h-2.5 w-2.5 border border-dashed border-smoke/60" />
   603→   603→          LIBRE
   604→   604→        </span>
   605→   605→        {showNow && (
   606→   606→          <span className="flex items-center gap-2 type-micro text-red">
   607→   607→            <span className="h-px w-4 bg-red" />
   608→   608→            AHORA · {minutesToTime(clock.minutes)}
   609→   609→          </span>
   610→   610→        )}
   611→   611→      </div>
   612→   612→
   613→   613→      {emptyDay && (

   638→   638→        {/* el canvas de slots */}
   639→   639→        <div className="relative ml-14 md:ml-16" role="list" aria-label={`Agenda de ${artist.name}`}>
   640→   640→          {slots.map((time) => {

   644→   644→              artist.bookings.some((b) => {

   646→   646→                return timeToMinutes(time) > bStart && timeToMinutes(time) < bStart + b.minutes;

   714→   714→            if (block) {
   715→   715→              const spanRows = Math.ceil(blockMinutes(block.minutes) / SHOP.slotStep);
   716→   716→              return (
   717→   717→                <button
   718→   718→                  key={time}
   719→   719→                  role="listitem"
   720→   720→                  disabled={busy}
   721→   721→                  onClick={() => {
   722→   722→                    if (confirmUnblock === block.id) {
   723→   723→                      onUnblock(block.id);
   724→   724→                      setConfirmUnblock(null);

   729→   729→                  }}
   730→   730→                  className="group absolute inset-x-0 px-4 py-2 text-left transition-colors"
   731→   731→                  style={{
   732→   732→                    height: spanRows * ROW_H - 4,
   733→   733→                    top: slots.indexOf(time) * ROW_H + 2,
   734→   734→                    background:
   735→   735→                      "repeating-linear-gradient(-45deg, rgba(229,35,27,0.10) 0 6px, rgba(229,35,27,0.03) 6px 12px)",
   736→   736→                    border: "1px solid rgba(229,35,27,0.35)",

   738→   738→                  aria-label={`Bloqueado ${time} a ${minutesToTime(

   751→   751→                  {spanRows > 1 && (
